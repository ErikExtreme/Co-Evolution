// In your Weapon or weapon controller:

public class Weapon : MonoBehaviour
{
    // Calculate how much grid/drone/power support we received
    float CalculateOffensiveSynergy()
    {
        // Grid synergy: how much available grid space enabled this weapon
        float gridUse = weaponGridSize / maxAvailableGridSize;
        
        // Drone synergy: how much drone support we have
        float droneSupport = activeDroneCount / maxDroneCount;
        
        // Power synergy: ratio of available power to what we use
        float powerAvailable = ship.Power / ship.PowerCapacity;

        return Mathf.Clamp01(synergy);
    }

    {
        float synergy = CalculateOffensiveSynergy();
        foreach (var moduleId in GetOffensiveSupportModuleIds())

                BattleMetricsRecorder.Instance.RegisterModuleOffensiveSynergy(moduleId, synergy);

void EndWave(bool playerSurvived)
{

            BattleMetricsRecorder.Instance.RegisterModuleRoleFulfillment(

                intendedMapping,
                actualMapping
            );
        }
    }
    
    // Register battle outcome
    ModuleManager.Instance.RegisterBattleMetricsForAll(battleDuration, playerSurvived);
}

Vector3 CalculateActualPerformanceMapping(ModuleStatsTracker tracker)
{
    // Based on what the module actually achieved, map it to 3D space
    // X = Durability contribution (HP, armor, shields provided)
    // Y = Weapon platform (grid, power, drones enabled)
    // Z = Stability (resource efficiency, consistency)
    
    float durability = track.damageEfficiency; // 0-1
    float weaponPlatform = tracker.offensiveSynergy; // 0-1
    float stability = tracker.powerEfficiency; // 0-1
    
    // Normalize to [-1, 1] range for each axis
    float x = (durability * 2f) - 1f;
    float y = (weaponPlatform * 2f) - 1f;
    float z = (stability * 2f) - 1f;
    
    return new Vector3(x, y, z);
}
*/

            return;

        }

        // End metrics recording
        if (BattleMetricsRecorder.Instance != null)
            BattleMetricsRecorder.Instance.EndBattle(playerSurvived);

        // Now evolution can use these metrics for fitness evaluation
        // EvolutionManager.Evolve() will calculate new fitness scores
    }
}

// ============================================================================
// NORMALIZATION REFERENCE
// ============================================================================
// When you register metrics, consider typical ranges for your game:
// 
// damageEfficiency: damage prevented per stat point
//   Typical range: 0-100+ depending on difficulty
//   Normalize with: value / 50f (adjust based on your balancing)
//
// survivalContribution: fraction of battle ship stayed alive
//   Range: [0, 1] - naturally normalized
//   Use directly in fitness calculation
//
// offensiveSynergy: how much modules enabled weapons
//   Range: [0, 1] - constrained by RegisterOffensiveSynergy clamping
//   Use directly in fitness calculation
//
// powerEfficiency: power available / power consumed
//   Typical range: 0.5 to 3.0 (higher = better efficiency)
//   Normalize with: Mathf.Clamp01(value / 2.0f)
//
// roleFulfillment: already normalized [0, 1] by RegisterRoleFulfillment
//   No additional normalization needed
//
// survivalSuccess: 1.0 = survived, 0.0 = died
//   Range: [0, 1] - use directly
